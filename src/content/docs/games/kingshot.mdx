## Want to help?

Drop into the **`#install`** channel on [Discord](https://discord.gg/62twnzKG9) and ping us. You don't need prior contribution experience — if you can describe Kingshot's daily loop in plain English, you can describe it in the DSL.

We have public docs for the full authoring flow already:

- [Authoring flow](/autopilot-page/authoring/flow/) — capture → label → write DSL → test
- [Labeling regions](/autopilot-page/authoring/labeling/) — using the editor
- [DSL primer](/autopilot-page/authoring/dsl/) — `match`, `click`, `while_match`, `cond`, `push_scenario` with a worked example

We'll pair on the labeling pass and review your first scenarios. There is now enough Kingshot structure in the repo to copy patterns instead of starting from an empty game tree.

The [Whiteout Survival coverage](/autopilot-page/games/whiteout-survival/) is still the template for what "fully covered" looks like — Kingshot is now on that path, just much earlier in the curve.

{/* Card-deck enhancement: turn the .scn-cards grid into the same 3D stack used
    on the Whiteout Survival page (hover / focus / tap to fan open). Self-contained
    — reuses the global .scn-stack styles from src/styles/wos.css. */}
<script is:inline>{`
(function () {
  function init() {
    var grids = document.querySelectorAll('.scn-cards');
    Array.prototype.forEach.call(grids, function (grid) {
      var cards = Array.prototype.filter.call(grid.children, function (c) {
        return c.classList && c.classList.contains('scn-card');
      });
      if (cards.length >= 2) makeStack(grid, cards);
    });
    initClipboard();
  }

  function makeStack(grid, cards) {
    grid.classList.add('scn-stack');
    grid.style.setProperty('--n', cards.length);
    cards.forEach(function (card, i) { card.style.setProperty('--i', i + 1); });
    grid.setAttribute('tabindex', '0');
    grid.setAttribute('role', 'group');
    grid.setAttribute('aria-label', 'Module deck — expand to see all ' + cards.length);
    grid.setAttribute('aria-expanded', 'false');

    requestAnimationFrame(function () {
      requestAnimationFrame(function () { grid.classList.add('scn-ready'); });
    });

    function setOpen(open) {
      grid.classList.toggle('is-open', open);
      grid.setAttribute('aria-expanded', open ? 'true' : 'false');
    }
    grid.addEventListener('click', function (e) {
      if (e.target.closest('code')) { setOpen(true); return; }
      setOpen(!grid.classList.contains('is-open'));
    });
    grid.addEventListener('keydown', function (e) {
      if (e.key === 'Enter' || e.key === ' ') {
        e.preventDefault();
        setOpen(!grid.classList.contains('is-open'));
      }
    });
  }

  var toastEl;
  function showToast(text) {
    if (!toastEl) {
      toastEl = document.createElement('div');
      toastEl.className = 'scn-toast';
      toastEl.setAttribute('role', 'status');
      document.body.appendChild(toastEl);
    }
    toastEl.textContent = text;
    toastEl.classList.add('is-shown');
    clearTimeout(toastEl._t);
    toastEl._t = setTimeout(function () { toastEl.classList.remove('is-shown'); }, 1500);
  }
  function initClipboard() {
    document.querySelectorAll('.scn-card code').forEach(function (c) {
      c.title = 'Click to copy';
      c.addEventListener('click', function () {
        var t = (c.textContent || '').trim();
        if (!t || !navigator.clipboard) return;
        navigator.clipboard.writeText(t).then(function () { showToast('Copied ' + t); });
      });
    });
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', init);
  } else {
    init();
  }
})();
`}</script>
