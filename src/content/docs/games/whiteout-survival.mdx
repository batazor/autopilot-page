    var reduceMotion = window.matchMedia
      && window.matchMedia('(prefers-reduced-motion: reduce)').matches;

    var dots = slides.map(function (_, i) {
      var dot = document.createElement('button');
      dot.type = 'button';
      dot.className = 'scn-slideshow-dot';
      dot.setAttribute('role', 'tab');
      dot.setAttribute('aria-label', 'Slide ' + (i + 1) + ' of ' + slides.length);
      dot.style.setProperty('--scn-autoplay-ms', AUTOPLAY_MS + 'ms');
      dot.appendChild(document.createElement('span'));
      dot.addEventListener('click', function () { goTo(i); restart(); });
      dotsHost.appendChild(dot);
      return dot;
    });

    function render() {
      slides.forEach(function (slide, i) {
        var active = i === current;
        slide.classList.toggle('is-active', active);
        slide.setAttribute('aria-hidden', active ? 'false' : 'true');
        // Keep links in hidden slides out of the tab order.
        slide.querySelectorAll('a').forEach(function (a) {
          if (active) a.removeAttribute('tabindex'); else a.tabIndex = -1;
        });
      });
      dots.forEach(function (dot, i) {
        var active = i === current;
        dot.classList.toggle('is-active', active);
        dot.classList.toggle('is-ticking', active && !paused && !reduceMotion);
        dot.setAttribute('aria-selected', active ? 'true' : 'false');
      });
    }

    function goTo(i) {
      current = (i + slides.length) % slides.length;
      render();
    }

    function restart() {
      clearInterval(timer);
      timer = setInterval(function () { goTo(current + 1); }, AUTOPLAY_MS);
      // Re-trigger the dot progress animation from zero.
      dots.forEach(function (dot) { dot.classList.remove('is-ticking'); });
      void dotsHost.offsetWidth;
      render();
    }

    function setPaused(value) {
      if (paused === value) return;
      paused = value;
      if (paused) clearInterval(timer); else restart();
      render();
    }

    root.querySelector('[data-slide-prev]').addEventListener('click', function () { goTo(current - 1); restart(); });
    root.querySelector('[data-slide-next]').addEventListener('click', function () { goTo(current + 1); restart(); });

    // Autoplay pauses while the reader is engaging with the card (or tab hidden).
    root.addEventListener('mouseenter', function () { setPaused(true); });
    root.addEventListener('mouseleave', function () { setPaused(false); });
    root.addEventListener('focusin', function () { setPaused(true); });
    root.addEventListener('focusout', function (e) {
      if (!root.contains(e.relatedTarget)) setPaused(false);
    });
    document.addEventListener('visibilitychange', function () {
      if (document.hidden) { clearInterval(timer); }
      else if (!paused) { restart(); }
    });

    // Keyboard arrows while the carousel has focus.
    root.addEventListener('keydown', function (e) {
      if (e.key === 'ArrowLeft') { e.preventDefault(); goTo(current - 1); restart(); }
      if (e.key === 'ArrowRight') { e.preventDefault(); goTo(current + 1); restart(); }
    });

    // Touch swipe with drag feedback: the active slide follows the finger
    // (touch-action: pan-y leaves vertical scrolling to the browser; a
    // vertical pan fires pointercancel and we snap back). On release past
    // the threshold the deck advances; otherwise the slide springs back.
    var drag = { id: null, startX: 0, dx: 0 };
    function activeSlide() { return slides[current]; }
    function dragReset(advanceTo) {
      root.classList.remove('is-dragging');
      slides.forEach(function (s) { s.style.transform = ''; s.style.opacity = ''; });
      drag.id = null;
      if (advanceTo !== undefined) goTo(advanceTo);
      restart();
    }
    root.addEventListener('pointerdown', function (e) {
      if (e.pointerType === 'mouse' || drag.id !== null) return;
      drag.id = e.pointerId;
      drag.startX = e.clientX;
      drag.dx = 0;
      try { root.setPointerCapture(e.pointerId); } catch (err) { /* no-op */ }
      clearInterval(timer);
    });
    root.addEventListener('pointermove', function (e) {
      if (e.pointerId !== drag.id) return;
      drag.dx = e.clientX - drag.startX;
      if (Math.abs(drag.dx) > 8) root.classList.add('is-dragging');
      var w = root.offsetWidth || 1;
      var slide = activeSlide();
      slide.style.transform = 'translateX(' + drag.dx + 'px)';
      slide.style.opacity = String(Math.max(0.35, 1 - Math.abs(drag.dx) / w));
    });
    root.addEventListener('pointerup', function (e) {
      if (e.pointerId !== drag.id) return;
      var w = root.offsetWidth || 1;
      var past = Math.abs(drag.dx) > Math.max(60, w * 0.15);
      dragReset(past ? current + (drag.dx < 0 ? 1 : -1) : undefined);
    });
    root.addEventListener('pointercancel', function (e) {
      if (e.pointerId === drag.id) dragReset();
    });

    if (reduceMotion) { render(); } else { restart(); }
  }

  // ---- Toast (shared) ----
  var toastEl;
  function showToast(text) {
    if (!toastEl) {
      toastEl = document.createElement('div');
      toastEl.className = 'scn-toast';
      toastEl.setAttribute('role', 'status');
      toastEl.setAttribute('aria-live', 'polite');
      document.body.appendChild(toastEl);
    }
    toastEl.textContent = text;
    toastEl.classList.add('is-shown');
    clearTimeout(toastEl._t);
    toastEl._t = setTimeout(function () { toastEl.classList.remove('is-shown'); }, 1500);
  }

  // ---- Highlighting helpers ----
  function clearMarks(el) {
    var marks = el.querySelectorAll('mark.scn-mark');
    marks.forEach(function (m) { m.replaceWith(document.createTextNode(m.textContent)); });
    if (marks.length) el.normalize();
  }
  function markMatches(el, q) {
    var walker = document.createTreeWalker(el, NodeFilter.SHOW_TEXT, {
      acceptNode: function (n) {
        if (!n.nodeValue || !n.nodeValue.trim()) return NodeFilter.FILTER_REJECT;
        var p = n.parentElement;
        if (!p || p.closest('script,style,mark,kbd')) return NodeFilter.FILTER_REJECT;
        return n.nodeValue.toLowerCase().indexOf(q) !== -1
          ? NodeFilter.FILTER_ACCEPT : NodeFilter.FILTER_REJECT;
      }
    });
    var nodes = [];
    while (walker.nextNode()) nodes.push(walker.currentNode);
    nodes.forEach(function (node) {
      var text = node.nodeValue;
      var lower = text.toLowerCase();
      var frag = document.createDocumentFragment();
      var idx = lower.indexOf(q), last = 0;
      while (idx !== -1) {
        if (idx > last) frag.appendChild(document.createTextNode(text.slice(last, idx)));
        var m = document.createElement('mark');
        m.className = 'scn-mark';
        m.textContent = text.slice(idx, idx + q.length);
        frag.appendChild(m);
        last = idx + q.length;
        idx = lower.indexOf(q, last);
      }
      if (last < text.length) frag.appendChild(document.createTextNode(text.slice(last)));
      node.parentNode.replaceChild(frag, node);
    });
  }

  // ---- Filter (count + mark) ----
  function initFilter(sections, totalItems) {
    var input = document.getElementById('scenario-filter');
    if (!input) return;
    var empty = document.getElementById('scenario-filter-empty');
    var count = document.getElementById('scenario-filter-count');
    var kbd = document.getElementById('scenario-filter-kbd');

    function apply() {
      var q = input.value.trim().toLowerCase();
      var visibleTotal = 0;
      sections.forEach(function (s) {
        var visible = 0;
        s.items.forEach(function (el) {
          clearMarks(el);
          var match = !q || (el.textContent || '').toLowerCase().indexOf(q) !== -1;
          el.hidden = !match;
          if (match) {
            visible++;
            if (q) markMatches(el, q);
          }
        });
        var sectionHidden = s.items.length > 0 && visible === 0;
        s.heading.hidden = sectionHidden;
        s.blocks.forEach(function (b) { b.hidden = sectionHidden; });
        visibleTotal += visible;
      });
      if (empty) empty.hidden = visibleTotal > 0 || !q;
      if (count && kbd) {
        if (q) {
          count.textContent = 'Showing ' + visibleTotal + ' of ' + totalItems + ' items';
          count.hidden = false;
          kbd.hidden = true;
        } else {
          count.hidden = true;
          kbd.hidden = false;
        }
      }
    }
    input.addEventListener('input', apply);
  }

  // ---- Scroll spy ----
  function initScrollSpy(sections, nav) {
    if (!('IntersectionObserver' in window)) return;
    var chips = nav.querySelectorAll('a[data-jump]');
    var setActive = function (id) {
      chips.forEach(function (c) {
        c.classList.toggle('scn-chip-active', c.getAttribute('data-jump') === id);
      });
    };
    var visible = new Set();
    var observer = new IntersectionObserver(function (entries) {
      entries.forEach(function (e) {
        if (e.isIntersecting) visible.add(e.target.id); else visible.delete(e.target.id);
      });
      for (var i = 0; i < sections.length; i++) {
        if (visible.has(sections[i].heading.id)) { setActive(sections[i].heading.id); return; }
      }
    }, { rootMargin: '-20% 0px -65% 0px', threshold: 0 });
    sections.forEach(function (s) { observer.observe(s.heading); });
  }

  // ---- Click-to-copy on scenario name pills ----
  function initClipboard(container) {
    var codes = container.querySelectorAll('td:first-child code');
    codes.forEach(function (c) {
      c.title = 'Click to copy';
      c.addEventListener('click', function () {
        var text = (c.textContent || '').trim();
        if (!text || !navigator.clipboard) return;
        navigator.clipboard.writeText(text).then(function () {
          showToast('Copied ' + text);
        });
      });
    });
  }

  // ---- Hash-link icon shown on heading hover ----
  function initAnchorLinks(container) {
    var headings = container.querySelectorAll('h2[id]');
    headings.forEach(function (h) {
      if (h.querySelector('.scn-anchor')) return;
      var a = document.createElement('a');
      a.href = '#' + h.id;
      a.className = 'scn-anchor';
      a.textContent = '#';
      a.setAttribute('aria-label', 'Copy link to ' + (h.textContent || '').trim());
      a.addEventListener('click', function (e) {
        e.preventDefault();
        var url = location.origin + location.pathname + '#' + h.id;
        history.replaceState(null, '', '#' + h.id);
        h.scrollIntoView({ behavior: 'smooth', block: 'start' });
        if (navigator.clipboard) {
          navigator.clipboard.writeText(url).then(function () { showToast('Link copied'); });
        }
      });
      h.appendChild(a);
    });
  }

  // ---- "/" focuses filter, Esc clears ----
  function initShortcut() {
    var input = document.getElementById('scenario-filter');
    if (!input) return;
    document.addEventListener('keydown', function (e) {
      var ae = document.activeElement;
      var inEditable = ae && (/^(input|textarea|select)$/i.test(ae.tagName) || ae.isContentEditable);
      if (e.key === '/' && !inEditable && !e.metaKey && !e.ctrlKey && !e.altKey) {
        e.preventDefault();
        input.focus();
        input.select();
      } else if (e.key === 'Escape' && ae === input) {
        input.value = '';
        input.dispatchEvent(new Event('input'));
        input.blur();
      }
    });
  }

  // ---- Reading-progress bar ----
  function initProgressBar() {
    var bar = document.getElementById('scn-progress');
    if (!bar) return;
    function update() {
      var doc = document.documentElement;
      var max = doc.scrollHeight - window.innerHeight;
      var pct = max > 0 ? (window.scrollY / max) * 100 : 0;
      bar.style.width = pct + '%';
    }
    window.addEventListener('scroll', update, { passive: true });
    window.addEventListener('resize', update);
    update();
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', init);
  } else {
    init();
  }
})();
