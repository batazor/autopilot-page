{/* WebGL backdrop: parallax snowfall + a per-theme sky show behind the hero.
    Dark theme: aurora borealis + bright additive snow. Light theme: a soft
    frost haze — slow layered mist with a cool tint, a veiled sun glow and a
    gentle glitter on the flakes. The u_lt uniform crossfades between the two
    looks in-shader when the theme switches.
    Degrades silently: no WebGL → the plain CSS gradient stays; reduced
    motion → a single static frame. Driven by the inline script below. */}
<canvas id="wos-aurora" aria-hidden="true"></canvas>

<script is:inline>{`
(function () {
  var canvas = document.getElementById('wos-aurora');
  if (!canvas) return;
  var gl = canvas.getContext('webgl', { alpha: true, premultipliedAlpha: true, antialias: false });
  if (!gl) { canvas.remove(); return; }

  var reduceMotion = window.matchMedia
    && window.matchMedia('(prefers-reduced-motion: reduce)').matches;

  function compile(type, src) {
    var s = gl.createShader(type);
    gl.shaderSource(s, src);
    gl.compileShader(s);
    if (!gl.getShaderParameter(s, gl.COMPILE_STATUS)) throw gl.getShaderInfoLog(s);
    return s;
  }
  function program(vs, fs) {
    var p = gl.createProgram();
    gl.attachShader(p, compile(gl.VERTEX_SHADER, vs));
    gl.attachShader(p, compile(gl.FRAGMENT_SHADER, fs));
    gl.linkProgram(p);
    if (!gl.getProgramParameter(p, gl.LINK_STATUS)) throw gl.getProgramInfoLog(p);
    return p;
  }

  var NOISE =
    'float hash(vec2 p){return fract(sin(dot(p,vec2(127.1,311.7)))*43758.5453123);}' +
    'float noise(vec2 p){vec2 i=floor(p),f=fract(p);f=f*f*(3.-2.*f);' +
    'return mix(mix(hash(i),hash(i+vec2(1,0)),f.x),mix(hash(i+vec2(0,1)),hash(i+vec2(1,1)),f.x),f.y);}' +
    'float fbm(vec2 p){float v=0.,a=.5;for(int k=0;k<4;k++){v+=a*noise(p);p*=2.03;a*=.5;}return v;}';

  // --- Sky pass: night aurora and daylight halo optics, mixed by u_lt ---
  var auroraPrg = program(
    'attribute vec2 a;void main(){gl_Position=vec4(a,0.,1.);}',
    'precision mediump float;uniform float u_t;uniform vec2 u_res;uniform vec2 u_par;uniform float u_lt;' + NOISE +
    'void main(){' +
    'vec2 uv=gl_FragCoord.xy/u_res;float t=u_t*.05;' +
    // Night: three noise-driven aurora curtains near the top
    'vec3 col=vec3(0.);' +
    'for(int i=0;i<3;i++){float fi=float(i);' +
    'float x=uv.x*1.6+fi*1.7+u_par.x*.05*(1.+fi*.5);' +
    'float wave=fbm(vec2(x*1.2+t*(.4+.25*fi),t*.7+fi*4.));' +
    'float y0=.62+.24*wave-fi*.07+u_par.y*.02;' +
    'float d=uv.y-y0;' +
    'float band=exp(-d*d*(d>0.?420.:160.));' +          // sharp crest, soft tail
    'float rays=.55+.45*fbm(vec2(uv.x*26.+fi*9.,t*2.2+uv.y*2.));' +
    'vec3 c=mix(vec3(.05,.85,.55),vec3(.25,.45,.95),fi*.5);' +
    'c=mix(c,vec3(.55,.3,.9),smoothstep(0.,.35,d)*.7);' +
    'col+=band*rays*c*(.5-fi*.11);}' +
    'col+=vec3(.03,.08,.14)*smoothstep(.45,1.,uv.y);' +  // faint night-sky glow
    'col*=.85;float colA=clamp(max(col.r,max(col.g,col.b)),0.,1.);' +
    // Day: soft frost haze — slow layered mist with a cool tint and a sun
    // veiled behind it. Layered alpha tints accumulated in premultiplied
    // form (dc = sum C*a); everything kept low-contrast on purpose.
    'vec2 asp=vec2(u_res.x/u_res.y,1.);' +
    'vec2 sun=vec2(.72+u_par.x*.012,.80+u_par.y*.012);' +
    'vec2 q=(uv-sun)*asp;float dd=length(q);' +
    'vec3 dc=vec3(0.);float da=0.;float ca;' +
    'float m1=fbm(vec2(uv.x*2.2-t*1.2+u_par.x*.03,uv.y*3.+t*.15));' +
    'float m2=fbm(vec2(uv.x*4.6+t*.8,uv.y*5.5-t*.12));' +
    'float mist=smoothstep(.32,.95,m1*.62+m2*.38);' +
    'ca=.14*mist*smoothstep(.5,1.,uv.y);dc+=vec3(.6,.72,.9)*ca;da+=ca;' +   // drifting top haze
    'ca=.10*mist*smoothstep(.5,0.,uv.y);dc+=vec3(.55,.68,.88)*ca;da+=ca;' + // ground drift
    'ca=.05*smoothstep(.2,1.,uv.y);dc+=vec3(.6,.73,.92)*ca;da+=ca;' +       // faint sky tint
    'ca=.10*exp(-dd*dd*7.);dc+=vec3(1.,.97,.9)*ca;da+=ca;' +                // veiled sun
    'da=clamp(da,0.,.6);dc=min(dc,vec3(da));' +
    // Crossfade the two premultiplied looks
    'gl_FragColor=vec4(mix(col,dc,u_lt),mix(colA,da,u_lt));}'
  );

  // --- Snow: GL points, three parallax depths, wrap-around fall ---
  var SNOW_N = 170;
  var snowPrg = program(
    'attribute vec3 a;uniform float u_t;uniform vec2 u_par;uniform float u_dpr;' +
    'varying float v_d;varying float v_tw;' +
    'void main(){v_d=a.z;' +
    'v_tw=.5+.5*sin(u_t*(2.+a.z*6.)+a.x*80.);' +         // per-flake glitter phase
    'float speed=mix(.012,.05,a.z);' +
    'float y=fract(a.y-u_t*speed);' +
    'float x=fract(a.x+sin(u_t*(.3+a.z)+a.y*40.)*.012*a.z+u_par.x*.015*a.z);' +
    'gl_Position=vec4(x*2.-1.,1.-y*2.,0.,1.);' +
    'gl_PointSize=mix(1.4,3.6,a.z)*u_dpr;}',
    'precision mediump float;varying float v_d;varying float v_tw;uniform float u_lt;' +
    'void main(){float r=length(gl_PointCoord-.5);' +
    'float a=smoothstep(.5,.12,r)*mix(.18,.5,v_d);' +
    'a*=mix(1.,.55+.55*v_tw,u_lt);' +                    // day: gentle glitter
    // night: additive white sparkle; day: slate-blue flakes that darken
    'vec3 c=mix(vec3(1.),vec3(.42,.52,.66),u_lt);' +
    'gl_FragColor=vec4(c*a,a);}'                         // premultiplied flake
  );

  // Geometry
  var quad = gl.createBuffer();
  gl.bindBuffer(gl.ARRAY_BUFFER, quad);
  gl.bufferData(gl.ARRAY_BUFFER, new Float32Array([-1, -1, 3, -1, -1, 3]), gl.STATIC_DRAW);
  var flakes = new Float32Array(SNOW_N * 3);
  for (var i = 0; i < SNOW_N; i++) {
    flakes[i * 3] = Math.random();
    flakes[i * 3 + 1] = Math.random();
    flakes[i * 3 + 2] = Math.pow(Math.random(), 1.6);   // few near flakes, many far
  }
  var snowBuf = gl.createBuffer();
  gl.bindBuffer(gl.ARRAY_BUFFER, snowBuf);
  gl.bufferData(gl.ARRAY_BUFFER, flakes, gl.STATIC_DRAW);

  var aLocA = gl.getAttribLocation(auroraPrg, 'a');
  var aLocS = gl.getAttribLocation(snowPrg, 'a');
  var uA = { t: gl.getUniformLocation(auroraPrg, 'u_t'), res: gl.getUniformLocation(auroraPrg, 'u_res'), par: gl.getUniformLocation(auroraPrg, 'u_par'), lt: gl.getUniformLocation(auroraPrg, 'u_lt') };
  var uS = { t: gl.getUniformLocation(snowPrg, 'u_t'), par: gl.getUniformLocation(snowPrg, 'u_par'), dpr: gl.getUniformLocation(snowPrg, 'u_dpr'), lt: gl.getUniformLocation(snowPrg, 'u_lt') };

  var dpr = 1, parX = 0, parY = 0, targX = 0, targY = 0;
  function resize() {
    // Self-healing: called every frame; cheap when nothing changed. Covers
    // the init race where the script runs before the stylesheet sizes the
    // canvas (clientWidth is still the 300x150 default at that point).
    dpr = Math.min(window.devicePixelRatio || 1, 1.25);  // glow doesn't need retina
    var w = Math.round(canvas.clientWidth * dpr);
    var h = Math.round(canvas.clientHeight * dpr);
    if (w === canvas.width && h === canvas.height) return;
    canvas.width = w;
    canvas.height = h;
    gl.viewport(0, 0, w, h);
  }

  function dark() { return document.documentElement.getAttribute('data-theme') !== 'light'; }
  var ltMix = dark() ? 0 : 1, ltTarg = ltMix;            // 0 = night look, 1 = day look

  function draw(ms) {
    resize();
    var t = ms * 0.001;
    parX += (targX - parX) * 0.04;                       // lazy parallax follow
    parY += (targY - parY) * 0.04;
    ltMix += (ltTarg - ltMix) * 0.06;                    // theme crossfade
    gl.clearColor(0, 0, 0, 0);
    gl.clear(gl.COLOR_BUFFER_BIT);
    gl.disable(gl.BLEND);
    gl.useProgram(auroraPrg);
    gl.bindBuffer(gl.ARRAY_BUFFER, quad);
    gl.enableVertexAttribArray(aLocA);
    gl.vertexAttribPointer(aLocA, 2, gl.FLOAT, false, 0, 0);
    gl.uniform1f(uA.t, t);
    gl.uniform2f(uA.res, canvas.width, canvas.height);
    gl.uniform2f(uA.par, parX, parY);
    gl.uniform1f(uA.lt, ltMix);
    gl.drawArrays(gl.TRIANGLES, 0, 3);
    gl.enable(gl.BLEND);
    gl.blendFunc(gl.ONE, gl.ONE);
    gl.useProgram(snowPrg);
    gl.bindBuffer(gl.ARRAY_BUFFER, snowBuf);
    gl.enableVertexAttribArray(aLocS);
    gl.vertexAttribPointer(aLocS, 3, gl.FLOAT, false, 0, 0);
    gl.uniform1f(uS.t, t);
    gl.uniform2f(uS.par, parX, parY);
    gl.uniform1f(uS.dpr, dpr);
    gl.uniform1f(uS.lt, ltMix);
    gl.drawArrays(gl.POINTS, 0, SNOW_N);
  }

  var running = false;
  function frame(ms) {
    if (!running) return;
    draw(ms);
    requestAnimationFrame(frame);
  }
  function setRunning(on) {
    if (on === running) return;
    running = on;
    if (on) requestAnimationFrame(frame);
  }
  function wanted() { return !document.hidden && !reduceMotion; }

  resize();
  window.addEventListener('resize', function () { resize(); if (!running) draw(12000); });
  document.addEventListener('visibilitychange', function () { setRunning(wanted()); });
  new MutationObserver(function () {
    ltTarg = dark() ? 0 : 1;
    if (reduceMotion) { ltMix = ltTarg; draw(12000); }   // re-render the static frame
  }).observe(document.documentElement, { attributes: true, attributeFilter: ['data-theme'] });
  window.addEventListener('pointermove', function (e) {
    targX = (e.clientX / window.innerWidth - 0.5) * 2;
    targY = (e.clientY / window.innerHeight - 0.5) * 2;
  }, { passive: true });

  canvas.classList.add('is-on');
  draw(12000);                                           // first frame right away,
  if (!reduceMotion) {                                   // even if the tab is hidden
    setRunning(wanted());
    // Backstops for loads in a hidden/prerendered tab and bfcache restores —
    // visibilitychange alone can be missed there.
    window.addEventListener('focus', function () { setRunning(wanted()); });
    window.addEventListener('pageshow', function () { setRunning(wanted()); });
  }
})();
`}</script>

## Plays the long game

Most bots just repeat a daily checklist. Autopilot also plans around *when* things happen and
*what* they cost:

<CardGrid>
  <Card title="Event-calendar look-ahead" icon="open-book">
    Reads the in-game event calendar (what's live today and what's coming over the next few days) and plans around it — holding resources for tonight's event instead of burning them this afternoon. Events are identical for everyone on a state, so one bot reads the calendar and the whole fleet on that state reuses the result.
  </Card>
  <Card title="Stamina budgeting" icon="setting">
    A per-account planner spreads regenerating stamina across competing uses (intel events, beast hunts, Joe bandit waves), reserves it for the highest-value event windows, and spends it down before the cap so none is wasted to overflow.
  </Card>
</CardGrid>
